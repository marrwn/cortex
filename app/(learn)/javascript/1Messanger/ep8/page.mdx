console.log("╚═══════════════════════════════════╝");
console.log(""); // empty line for spacing

// === Items (use math for each!) ===
// price per unit × quantity = line total
const rice = 15 * 2;       // 30 LE
const oil = 85 * 1;        // 85 LE  
const sugar = 15 * 3;      // 45 LE
const tea = 65 * 1;        // 65 LE
const bread = 1 * 5;       // 5 LE

// Display as a table
console.table([
  { item: "Rice (1kg)", qty: 2, price: rice + " LE" },
  { item: "Cooking Oil", qty: 1, price: oil + " LE" },
  { item: "Sugar (1kg)", qty: 3, price: sugar + " LE" },
  { item: "Tea (Pack)", qty: 1, price: tea + " LE" },
  { item: "Bread", qty: 5, price: bread + " LE" }
]);

// === Calculations ===
const subtotal = rice + oil + sugar + tea + bread;
const tax = subtotal * 0.14;
const total = subtotal + tax;

// === Print Totals ===
console.log("─────────────────────────────────");
console.log("Subtotal:  " + subtotal + " LE");
console.log("Tax (14%): " + tax + " LE");
console.log("Total:     " + total + " LE");
console.log("─────────────────────────────────");

// === Warnings ===
if (total > 500) {
  console.warn("⚠️ Big spender! Total exceeds 500 LE!");
}

console.log("\\nThank you for shopping with us! 🛒");`
